# hd.c (00000.c) is matched C; its NON_MATCHING variants of a few call sites pass
# the leaked registers the original hands its asm callees (traced in the original
# ROM, see the comment above veh_update_others).
# veh_update_others: the re-ground callees get fp = 0 (the game thread's $s8). The
# matching build's IDO code leaves fp alone, so with fp = 0 in the registers both
# builds agree; any other fp fails (the C passes the traced constant). a3 and the
# FP state for starski_reground are 0 as in the C (dead, see the C comment); with
# several records the asm hands the next callee the FP state the previous one
# left, so veh_reground's FP inputs aren't compared. The matrix-chain callees'
# inputs (only read for a count-0 point, never seen in the trace) aren't either.
veh_update_others: -n 100 --ret void --arg fp=0 --arg a3=0 --arg f12=0 --arg f14=0 --arg f20=0 --arg f22=0 --arg f24=0 --arg f26=0 \
    --mem gVehiclesEnd=rel:gVehicles+0x74 --mem gVehicles+0x70=1 --mem gCurVehicle:1=0x20 \
    --mem gVehicles+0x5C=choice:1,3,4,5,8,9,13,14,15 \
    --follow sideswipe_reground,ramdozer_reground,skyfall_reground,jbomb_reground,starski_reground,backlash_reground,amdream_reground,police_reground,ateam_reground \
    --sig debug_printf=a0 --sig amdream_build_model= --sig police_build_model= --sig ateam_build_model= --sig backlash_build_model= --sig starski_build_model= \
    --sig sideswipe_build_model= --sig ramdozer_build_model= --sig jbomb_build_model= --sig skyfall_build_model= \
    --sig veh_reground=a0,a1,a2,a3,stack0,stack1,stack2,stack3,stack4,stack5
# three records: skipped ones (flag 0, the current vehicle, 0, 0xFE, 0xFF), the
# default case (debug print) and every handled type

    --mem gVehiclesEnd=rel:gVehicles+0x15C --mem gVehicles+0x70*0x74:3=choice:0,1*3 --mem gCurVehicle:1=choice:0x20,3 \
