fx_finish_all: -n 300 --ret void --mem gEffects:0x3C0=words:0x80B00000,0x80B00020,0x80B00040,0x80B000F0 --mem 0x80B00000:0x100=rand

# fx_alloc_texture_caches: carve 0x1010-byte blocks from the heap pointer up to gFxHeapEndKind20 (flag 0x20) or

# tex_decode_next: tex_decode(cursor gTexDecodeQueueHead), then cursor += 0x10. The asm relies on
# tex_decode preserving a0 AND v1 (a stub clobbers v1 and the reference faults), so the
# callee chain is run for real (--follow-all). Record at heap0: src = heap0+0x800, even size
# 0..0x3E (ld/sd copy of size&~7, then halves), type 0..6 (decoders) or none, t4 word.
# Decoder input halves: literals, and back-references with even offsets (odd ones fault on lhu).
# The table word (+0xC, read by types 4 and 5) must point at real table memory, not `int`: a
# random int can land in .hd_code_data on a code-pointer table (jump tables, callbacks), whose
# words legitimately differ between the builds (0x802Dxxxx vs the relocated 0x8089xxxx), so
# the decoded colours differ although both decoders are right (seen as 9 bytes at heap+0x804).

    --mem @heap0+0x800:0x40=halves:0x1234,0x0042,0x7FFF,0x8041,0x8443,0x8C5F,0x0000 --mem gTexDecodeScratch:0x40=rand
