# 7FB50: status bitstream packers over the 0xFC-byte record table gObjects
# (gObjectsEnd = one past the last record in use; must be on the stride), and
# jbomb_track_lr_taps (two press/release trackers).

# record_status (record_status): bit per part (== 100), then rdu_is_lit(i)
# for i < gRduCount. Whole table filled from a byte pool (numParts at +0xE9
# up to 100, which deliberately runs past the record into the next one).
record_status: -n 300 --sig rdu_is_lit=a0 --arg a0=ptrz:0x800 \
    --mem gObjects:0x3624=bytes:0,1,2,3,7,8,9,100*3,99,101 \
    --mem gObjectsEnd=rel:gObjects+0xFC*int:0:10 --mem gRduCount:2=int:0:40
# full table (55 records), count boundaries around a byte
record_status: -n 100 --seed 1 --sig rdu_is_lit=a0 --arg a0=ptrz:0x1000 \
    --mem gObjects:0x3624=bytes:0,1,2,8,100*2 \
    --mem gObjectsEnd=rel:gObjects+0x3624 --mem gRduCount:2=choice:0,1,7,8,9,15,16,17,0x100
# empty table, no calls / few calls
record_status: -n 30 --seed 2 --sig rdu_is_lit=a0 --arg a0=ptrz:0x100 \
    --mem gObjectsEnd=sym:gObjects --mem gRduCount:2=choice:0,1,8
# numParts 0xFF / 0 in exactly one record (others small)
record_status: -n 100 --seed 3 --sig rdu_is_lit=a0 --arg a0=ptrz:0x800 \
    --mem gObjects:0x3624=bytes:1,2,100 \
    --mem gObjects:0x9D8=onehot:0xFC@0xE9/1:0,0xFF:bytes:1,2,100 \
    --mem gObjectsEnd=rel:gObjects+0x9D8 --mem gRduCount:2=int:0:20
